package MemberCrossFeatures {
    classifier C1 {
        feature d : C2;
    }
    classifier C2;
    assoc Base {
        end feature x : C1 {
            member feature xc : C1;
        }
        end feature y : C2 {
            member feature yc : C2;
        }
        end feature z : C2;
    }
    assoc Derived specializes Base {
        // Conforming: the crossed chain's second feature is the owned cross feature xc2, which
        // implicitly subsets Base::x's crossFeature xc.
        end feature x : C1 crosses x::p.xc2 {
            member feature xc2 : C1 featured by C2;
            member feature p : C2 featured by Derived {
                public import xc2;
            }
        }
        // Invalid: d does not specialize Base::y's crossFeature yc.
        end feature y : C2 crosses x.d;
        end feature z : C2;
    }
}
