# SOURCE
~~~kerml
package Diamond {
    classifier Value;
    behavior A { in a : Value; }
    behavior B specializes A { in b : Value; }
    behavior C specializes A;
    // D's parameters are [B::b] alone: A::a is redefined by B::b, so it is not inherited via C.
    behavior D specializes B, C;
    // E's first parameter pairs with B::b; its second pairs with nothing.
    behavior E specializes D { in e1 : Value; in e2 : Value; }

    classifier P;
    assoc EndA { end feature ea1 : P; end feature ea2 : P; }
    assoc EndB :> EndA { end feature eb : P; }
    assoc EndC :> EndA;
    // EndD's endFeature is [EndB::eb, EndA::ea2]: EndA::ea1 is redefined by EndB::eb.
    assoc EndD :> EndB, EndC;
    // The ends pair with EndB::eb and EndA::ea2; the third pairs with nothing.
    assoc EndE :> EndD { end feature ee1 : P; end feature ee2 : P; end feature ee3 : P; }
}
~~~
