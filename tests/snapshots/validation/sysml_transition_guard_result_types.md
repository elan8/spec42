~~~sysml
package Guards {
    private import ScalarValues::*;
    part def Controller {
        attribute count : Integer;
        attribute ready : Boolean;
        attribute anything;
        state def Machine {
            entry; then idle;
            state idle;
            state busy;

            // Conforming: a Boolean feature, a comparison, a logical operator.
            transition flag first idle if ready then busy;
            transition compare first idle if count > 0 then busy;
            transition logical first idle if ready and count > 0 then busy;

            // Undecided, not reported: an untyped referent has the type of its value.
            transition open first idle if anything then busy;

            // Invalid: an Integer feature, and Integer arithmetic.
            transition integer first busy if count then idle;
            transition arithmetic first busy if count + 1 then idle;
        }
    }
}
~~~
