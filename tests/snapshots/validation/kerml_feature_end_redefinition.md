# SOURCE
~~~kerml
package Redefinition {
    classifier A;
    classifier B;
    assoc Parent { end feature a : A; end feature b : B; }
    // Each owned end redefines the parent's end at its position.
    assoc Child :> Parent { end feature x : A; end feature y : B; }
    // One owned end redefines the first parent end; the second is inherited.
    assoc Partial :> Parent { end feature only : A; }
    // The second position pairs with the end Partial inherits, not with an end it owns.
    assoc Grand :> Partial { end feature g1 : A; end feature g2 : B; }
    // An authored redefinition does not replace the positional one.
    assoc Swapped :> Parent { end feature s1 :>> Parent::b; end feature s2 :>> Parent::a; }
}
~~~
