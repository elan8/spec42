# SOURCE
~~~sysml
// Every StateActionUsage form is the member of a StateSubactionMembership and redefines the
// StateAction feature of its kind: the EmptyActionUsage (`entry;`), the declared action, the
// reference to an existing action, and an effect form. The assignment's operands are not lowered
// (unsupported), but its occupancy of the entry membership is.
package StateActions {
    action def Work;
    state def Bare {
        entry;
        do action work : Work;
    }
    state def Referencing {
        action shutdown : Work;
        exit shutdown;
    }
    state def Effects {
        attribute count;
        entry assign count := 0;
    }
}
~~~
