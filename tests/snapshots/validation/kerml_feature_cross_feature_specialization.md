package Crossings {
    classifier C1 {
        feature a : C2;
        feature d : C2;
    }
    classifier C2 {
        feature b : C1;
        feature c : C1 subsets b;
    }
    assoc Base {
        end feature x : C1 crosses y.b;
        end feature y : C2 crosses x.a;
    }
    assoc Derived specializes Base {
        // Conforming: c subsets Base::x's crossFeature b.
        end feature x : C1 crosses y.c;
        // Invalid: d does not specialize Base::y's crossFeature a.
        end feature y : C2 crosses x.d;
    }
}
