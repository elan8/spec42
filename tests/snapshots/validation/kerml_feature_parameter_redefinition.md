# SOURCE
~~~kerml
package Parameters {
    classifier Value;
    behavior Base { in a : Value; in b : Value; out c : Value; }
    // Each owned parameter redefines the parameter at its position of the general Behavior.
    behavior Positional specializes Base { in x : Value; in y : Value; out z : Value; }
    // One owned parameter redefines the first; the rest are inherited.
    behavior Partial specializes Base { in only : Value; }
    // The second position pairs with the parameter Partial inherits, not with one it owns.
    behavior Grand specializes Partial { in g1 : Value; in g2 : Value; }
    // A Step's parameters pair with the parameters of the Behavior typing it.
    step performed : Base { in s1 : Value; }
    // A result is no position: the parameter after it still pairs with the second one.
    function Computing { in p : Value; return r : Value; in q : Value; }
    function Refining specializes Computing { in p2 : Value; in q2 : Value; return r2 : Value; }
    // A positional invocation argument redefines the callee's parameter at its position; a named
    // argument owns the redefinition it names instead.
    feature computed = Computing(1, 2);
    feature named = Computing(q = 3);
}
~~~
