# SOURCE
~~~sysml
package Redefinition {
    requirement def Need;
    case def Base { objective baseObjective : Need; }
    case def Derived :> Base { objective derivedObjective : Need; }
    // No objective of its own: Inheriting's objectiveRequirement is the one Derived owns.
    case def Inheriting :> Derived;
    case def Grand :> Inheriting { objective grandObjective : Need; }
    // A case usage's objective redefines the objective of the case definition typing it.
    case run : Base { objective runObjective : Need; }
}
~~~
