~~~sysml
package Metadata {
    private import ParametersOfInterestMetadata::*;

    part def Wheel;

    // A metadata definition restricts what it may annotate by specializing annotatedElement.
    // A member with no kind keyword is a reference usage, not an attribute, so its only types
    // are the metaclass it names and what annotatedElement already has.
    metadata def Subsetted { :> annotatedElement : SysML::RequirementUsage; }
    metadata def Redefined { :>> annotatedElement : SysML::RequirementUsage; }
    metadata def Referenced { ref :>> annotatedElement : SysML::RequirementUsage; }

    // Conforming: each annotated element is a requirement usage.
    requirement tagged { @Subsetted; @Redefined; @Referenced; }
    requirement other;
    metadata onRequirement : Subsetted about other;

    // Conforming: a standard library definition applied to an element it admits.
    attribute mass : ScalarValues::Real { @MeasureOfPerformance; }

    // Invalid: the annotated element is a part usage, which Subsetted does not admit.
    part wheel : Wheel { @Subsetted; }
}
~~~
