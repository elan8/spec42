package Constructors {
    classifier Value;
    classifier Pair { feature first : Value; feature second : Value; }
    classifier Triple specializes Pair { feature third : Value; }
    classifier Holder {
        // Each positional argument redefines the public feature at its position.
        feature pair = new Pair(1, 2);
        // A named argument redefines the feature it names.
        feature named = new Pair(second = 3);
        // A type's owned features come before the ones it inherits.
        feature triple = new Triple(4, 5, 6);
        // A nested constructor is its own expression, whose result owns its own arguments.
        feature nested = new Pair(new Pair(7, 8), 9);
    }
}
