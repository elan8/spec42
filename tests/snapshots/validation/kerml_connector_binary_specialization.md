~~~kerml
package Connectors {
    classifier Thing;
    classifier Holder {
        feature a : Thing;
        feature b : Thing;
        feature c : Thing;

        // Conforming: a two-ended connector typed by BinaryLink.
        // FeatureDeclaration uses TypedBy (`:` / `typed by`), not classifier `specializes`.
        connector pair : Links::BinaryLink { end feature e1 :>> a; end feature e2 :>> b; }

        // Invalid: a three-ended connector must not specialize BinaryLink.
        connector tern : Links::BinaryLink { end feature e1 :>> a; end feature e2 :>> b; end feature e3 :>> c; }
    }
}
~~~
