package Remaining {
    abstract classifier Anything;
    abstract class Occurrence specializes Anything {
        feature startShot;
        feature endShot;
    }
    assoc BinaryLink {
        end feature source: Occurrence;
        end feature target: Occurrence;
    }
    assoc struct BinaryLinkObject specializes BinaryLink;
    interaction Transfer specializes BinaryLink {
        end feature source: Occurrence redefines BinaryLink::source {
            feature sourceOutput: Anything[0..*];
        }
        end feature target: Occurrence redefines BinaryLink::target {
            feature targetInput: Anything[0..*];
        }
        feature payload: Anything[1..*];
        connector sourceOutputLink: BinaryLinkObject {
            end feature transferSource references source;
            end feature transferPayload references payload subsets transferSource.sourceOutput;
        }
        connector targetInputLink: BinaryLinkObject {
            end feature transferTarget references target;
            end feature transferPayload references payload subsets transferTarget.targetInput;
        }
    }
}
