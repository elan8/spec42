package Kinds {
    part def Chassis;
    item def Cargo;
    action def Braking;
    enum def Colour { enum red; enum blue; }
    attribute def Weight;
    attribute def Kilos :> Weight;
    port def Data;

    part def Holder {
        // Accepted. A part usage typed by a part definition, and by an item definition -- a part
        // definition specializes an item definition, so the two families are comparable.
        part frame : Chassis;
        part load : Cargo;

        // Rejected. An action definition is in the behaviour branch and a port definition in
        // neither, so neither is comparable with the part family.
        part stop : Braking;
        part socket : Data;

        // Accepted: an attribute usage typed by an enumeration definition, which specializes the
        // attribute family. Rejected: typed by a part definition.
        attribute shade : Colour;
        attribute weight : Chassis;
    }

    // An analysis case is a case, which is an action; a part definition is not in that branch, so
    // it cannot type an analysis usage. Its own family and its ancestors' can.
    analysis def Study;
    analysis rejected : Chassis;
    analysis accepted : Study;

    // Accepted, then rejected: a definition may specialize its own family and the families it
    // descends from, but not one in an unrelated branch.
    part def Trailer :> Chassis;
    part def Wagon :> Cargo;
    part def Brake :> Braking;

    part def Fleet {
        part vehicles[2..4] : Chassis;
    }

    // Each of these redefines `vehicles` implicitly, by redeclaring its name. Narrowing both
    // bounds is accepted; widening either one is not.
    part def Narrowed :> Fleet {
        part vehicles[3..3] : Chassis;
    }
    part def WidenedLower :> Fleet {
        part vehicles[1..4] : Chassis;
    }
    part def WidenedUpper :> Fleet {
        part vehicles[2..9] : Chassis;
    }
    part def WidenedUnbounded :> Fleet {
        part vehicles[2..*] : Chassis;
    }

    // A redefinition must keep a conforming type. `Trailer` specializes `Chassis`, so narrowing
    // to it conforms; retyping to an unrelated definition does not.
    part def ConformingRetype :> Fleet {
        part vehicles[2..4] : Trailer;
    }
    part def UnrelatedRetype :> Fleet {
        part vehicles[2..4] : Cargo;
    }

    // Subsetting is judged by the same family relation, and by the same type rule as
    // redefinition. `spare` subsets a part usage; `misuse` subsets an action usage, whose family
    // is unrelated; `mistyped` subsets a part usage but retypes itself to an unrelated family.
    part def Yard {
        part kept : Chassis;
        action running : Braking;
        part spare :> kept;
        part misuse :> running;
        part mistyped : Cargo :> kept;
    }

    // Reference and cross subsetting are judged by the same family and type rules as plain
    // subsetting, but neither resolves in this publication's slice: both publish
    // `unsupported_reference`, and an unsupported prerequisite settles no conformance verdict on
    // top of it. The rules are reachable the moment the references resolve.
    part def Referencing {
        attribute narrow : Kilos;
        attribute widened : Weight references narrow;
        attribute crossed : Weight crosses narrow;
    }

    // A specialization cycle, reported once per declaration on it.
    part def Loop1 :> Loop2;
    part def Loop2 :> Loop1;
    part def SelfLoop :> SelfLoop;
}
