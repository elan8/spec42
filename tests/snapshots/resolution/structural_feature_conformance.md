package Structural {
    part def Pump;
    part def Tank;
    attribute def Reading;
    action def Pumping;

    // A flow payload carries an occurrence. A part definition is one; an attribute definition is a
    // value type and is not.
    action def Moving {
        action a;
        action b;
        flow of Pump from a to b;
        flow of Reading from a to b;
    }

    // Two ends is complete; one is an incomplete pair. A flow definition is binary, so three ends
    // is too many, while a connection definition may be n-ary.
    connection def Complete {
        end left : Pump;
        end right : Tank;
    }
    connection def Incomplete {
        end only : Pump;
    }
    connection def Ternary {
        end one : Pump;
        end two : Tank;
        end three : Pump;
    }
    flow def BinaryFlow {
        end source : Pump;
        end target : Tank;
        end extra : Pump;
    }

    // An abstract declaration is legitimately allowed an incomplete end set, and this one is
    // reported anyway: the pinned parser carries no abstractness field on any connection-like
    // definition, so the prefix is dropped before semantics sees it. See
    // planning/UPSTREAM_PARSER_GAPS.md, Gap 58 -- the limitation is shown here rather than hidden.
    abstract connection def AbstractIncomplete {
        end only : Pump;
    }

    // A redefinition is introduced by the featuring type of the redefined feature or one that
    // specializes it. `Derived` specializes `Base`, so its redefinition is well-formed; `Unrelated`
    // names the inherited feature through its owner and does not.
    part def Base {
        attribute reading : Reading;
    }
    part def Derived :> Base {
        attribute :>> reading : Reading;
    }
    part def Unrelated {
        attribute mine :>> Base::reading;
    }

    // A redefinition may keep or narrow a direction, but not contradict one. `inout` is the widest,
    // so narrowing it to `out` conforms; `in` and `out` contradict each other.
    action def Directions {
        in attribute strict : Reading;
        inout attribute widest : Reading;
    }
    action def Narrowing :> Directions {
        out attribute strict : Reading;
        out attribute widest : Reading;
    }

    // A non-unique feature cannot subset a unique one; the reverse is fine. Uniqueness is the KerML
    // default where no `nonunique` was authored, which is why the subsetted side needs no keyword.
    part def Uniqueness {
        attribute pool : Reading[0..*];
        attribute repeated : Reading[0..*] nonunique :> pool;
        attribute distinct : Reading[0..*] :> pool;
    }

    // A typed variant member uses the variation's own family.
    variation part def Choice {
        variant goodChoice;
        variant badChoice;
    }
    part goodChoice : Pump;
    action badChoice : Pumping;
}
