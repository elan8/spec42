# META
~~~ini
description=An alias is a non-owning membership whose name identifies its member element (SysML 7.5.2): every reference through an alias -- local, imported, re-exported by public imports, membership-imported or qualified, and for typing, specialization and subsetting alike -- resolves to the aliased element, following chains of aliases. A private alias is not visible outside its namespace, and a use of an alias whose own target does not resolve is unresolved without a second diagnostic.
type=file
~~~

    part t : Tyre;

    part y : Facade::Tyre;
    part def V8 :> Motor;
    part engine : Engine;
    alias MainEngine for engine;
    part spare :> MainEngine;
    part gone : Missing;
}

    part t : Tyre;
}

}
