package Contexts {
    attribute externalLimit = 400;
    part def Pump { attribute pressure; }
    part def SpecialPump :> Pump { attribute :>> pressure; }
    part def Vehicle {
        part pump : Pump;
        attribute mass;
        attribute limit = 100;
        constraint massBudget { mass <= limit }
        constraint localBudget { in attribute local; local <= limit }
        constraint externalBudget { mass <= externalLimit }
        constraint pressureBudget { pump.pressure <= limit }
    }
    part def Truck :> Vehicle {
        attribute :>> limit = 200;
        part :>> pump : SpecialPump;
        attribute cargo subsets mass;
    }
    part truck : Truck {
        attribute :>> limit = 300;
    }
    part def HeavyTruck :> Truck {
        constraint inheritedBudget redefines massBudget;
    }
    part def ConflictingHeavyTruck :> HeavyTruck {
        constraint firstBudget redefines inheritedBudget;
        constraint secondBudget redefines inheritedBudget;
    }
    part def CustomTruck :> Vehicle {
        constraint customBudget redefines massBudget { mass < limit }
    }
    part def ConflictingTruck :> Vehicle {
        attribute limitA redefines limit;
        attribute limitB redefines limit;
    }
}
