package SubjectTypes {
    part def Subsystem;
    part def System { part sub : Subsystem; }
    part def SpecialSystem :> System;
    requirement def SystemRequirement { subject system : System; }
    requirement def SpecialRequirement :> SystemRequirement {
        subject :>> system : SpecialSystem;
    }
    requirement requirement1 : SystemRequirement;
    requirement requirement2 : SpecialRequirement;
    part system1 : System;
    part system2 : SpecialSystem;
    satisfy requirement1 by system1;
    satisfy requirement1 by system2;
    satisfy requirement1 by system1.sub;
    satisfy requirement2 by system2;
    satisfy requirement2 by system1;
    satisfy requirement inlineValid : SystemRequirement by system1;
    satisfy requirement inlineInvalid : SystemRequirement by system1.sub;
    part def ConformingContext :> System { satisfy requirement1 by that; }
    part def IncompatibleContext { satisfy requirement1 by that; }
}
