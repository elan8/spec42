# META
~~~ini
description=`inspection().connections(root)` publishes every `connect` / `interface` connector reachable from an element with its type and resolved ends - a bare `connect a to b` end is a single feature, a dotted `connect a.b to c.d` end carries the resolved terminal plus the path as authored, and an end whose target does not resolve stays unresolved rather than dropped
type=file
~~~
# SOURCE
~~~sysml
package ConnectionTopology {

    port def PowerPort;

    part def Pump {
        port inlet : PowerPort;
        port outlet : PowerPort;
    }

    interface def PowerInterface {
        end supply : PowerPort;
        end draw : PowerPort;
    }

    // A `connection def` whose ends are declared but not wired: each is published as an
    // `unconnected` end carrying its own identity.
    connection def Feed {
        end from : PowerPort;
        end to : PowerPort;
    }

    // A `connection def` with a `:>` supertype (no `:` type, so `declared_type` stays absent).
    connection def RefinedFeed :> Feed;

    part system {
        part pumpA : Pump;
        part pumpB : Pump;

        // Bare ends: two single-feature connector ends.
        connect pumpA to pumpB;

        // Dotted ends: each carries the resolved terminal port plus the authored path.
        connect pumpA.outlet to pumpB.inlet;

        // Mixed: one bare end and one named end -- published in authored order.
        connect pumpA to keyed references pumpB.inlet;

        // Named ends (`connect x references ...`), each published with its own identity.
        connection feed : Feed
            connect supplyEnd references pumpA.outlet
                 to drawEnd references pumpB.inlet;

        // An interface usage typed by an interface def, with dotted ends.
        interface pumpLink : PowerInterface connect pumpA.outlet to pumpB.inlet;

        // An end whose target does not resolve.
        connect pumpA.outlet to missing.port;
    }
}
~~~
