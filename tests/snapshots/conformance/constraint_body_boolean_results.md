package Constraints {
    private import ScalarValues::*;
    part def Tank {
        attribute level : Integer;
        attribute full : Boolean;
        attribute anything;

        // Conforming.
        constraint isFull { full }
        constraint notEmpty { level > 0 }
        assert constraint consistent { full implies level > 0 }
        requirement bounded { require constraint { level <= 100 } }

        // Undecided, not reported: an untyped referent has the type of its value.
        constraint open { anything }

        // Invalid: Integer results.
        constraint raw { level }
        assert constraint sum { level + 1 }
        requirement scaled { require constraint { level * 2 } }
    }
}
