package Connections {
    port def Signal;
    port def Power;
    part def Load;

    part def Conforming {
        port left : Signal;
        port right : Signal;
        connect left to right;
    }

    part def MismatchedPortTypes {
        port left : Signal;
        port right : Power;
        connect left to right;
    }

    part def EndpointIsNotAPort {
        port left : Signal;
        part load : Load;
        connect left to load;
    }

    part def RepeatedPair {
        port left : Signal;
        port right : Signal;
        connect left to right;
        connect left to right;
    }

    part def NothingConnectsThisPort {
        port lonely : Signal;
    }

    attribute def Enabled;
    port def Directed {
        in enabled : Enabled;
    }

    part def ConformingDirections {
        port offering : Directed;
        port receiving : ~Directed;
        connect offering to receiving;
    }

    part def MirroredDirections {
        port left : ~Directed;
        port right : ~Directed;
        connect left to right;
    }
}
