# SOURCE
~~~sysml
package Specialization {
    part def Engine;
    part def Vehicle {
        part engine : Engine;
    }
    part engine : Engine;

    // Conforming: a definition specializes a definition, a usage subsets a usage.
    part def V8 :> Engine;
    part spare :> engine;
    part def Truck :> Vehicle {
        part bigEngine :>> engine;
    }

    // Invalid: a definition specializes a usage.
    part def V12 :> engine;

    // Invalid: a usage subsets, redefines, and references a definition.
    part wrong :> Engine;
    part def Car :> Vehicle {
        part motor :>> Engine;
    }
    ref part pointer ::> Engine;
}
~~~
