package Functions {
    action def Step;
    action stepA : Step;
    action stepB : Step;

    // Conforming: a function tree built from usages.
    action group {
        perform stepA;
        perform action second :> stepB;
        action nested { perform stepB; }
    }
    action def Group {
        perform stepA;
    }

    // Conforming: an action usage bound to a value.
    action def Torque { action generate; }
    action fourCylinder;
    part def Engine {
        perform action provide : Torque;
    }
    part engine : Engine {
        perform action :>> provide {
            action :>> generate = fourCylinder;
        }
        timeslice warmUp {
            perform stepA;
        }
    }

    // Conforming: an anonymous perform is named by the action it performs, so a specializing
    // part can redefine it, also when the action is named by a qualified name.
    package Tree {
        action providePower : Torque;
    }
    part family {
        perform Tree::providePower;
        perform stepA;
        perform stepA;
    }
    part variant :> family {
        perform action :>> providePower {
            action :>> generate = fourCylinder;
        }
    }

    // Conforming: loop nodes as succession targets.
    action def Run {
        attribute i : ScalarValues::Integer;
        first start;
        then while i > 0 {
            perform stepA;
        }
        then loop {
            perform stepB;
        } until i > 3;
        then done;
    }
}
