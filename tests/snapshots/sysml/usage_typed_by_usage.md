# SOURCE
~~~sysml
package Typing {
    private import ISQElectromagnetism::electricPower;

    part def Drone;
    part drone : Drone;
    attribute def Level;
    attribute level : Level;

    port def PowerPort {
        // Invalid: `electricPower` is an attribute usage in ISQ, not a type.
        out power : electricPower;

        // Conforming: the feature subsets the usage.
        out supplied :> electricPower;
    }

    // Invalid: an attribute and a part typed by usages.
    attribute copy : level;
    part clone : drone;

    // Conforming: typed by the definitions.
    attribute other : Level;
    part second : Drone;

    action def Loop {
        in items [0..*];

        // Conforming: `:>` on a `ref` in an action body is a subsetting.
        ref current [0..1] :> items;
    }
}
~~~
