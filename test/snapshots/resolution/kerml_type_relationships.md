package Sets {
    classifier Wheel;
    classifier Fork;
    classifier Frame;
    classifier Missing2;

    // Each clause publishes one relationship per target, in authored order. None of them is a
    // specialization, so none appears in the owner's supertypes.
    classifier Parts unions Wheel, Fork;
    classifier Shared intersects Wheel, Fork;

    // Differencing order is load-bearing: `Wheel` is the type being reduced, and the remaining
    // targets are exclusions. A repeated clause continues the same exclusion list rather than
    // starting a new one, so `Frame` takes ordinal 2.
    classifier Reduced differences Wheel, Fork differences Frame;

    // Interleaved clauses of different kinds keep separate ordinal sequences.
    classifier Mixed unions Wheel, Fork intersects Frame, Wheel unions Frame;

    // Disjointness is pairwise, so a single target is well-formed.
    classifier Apart disjoint from Wheel;

    // Rejected: a union, intersection or difference of one type is that type, so KerML requires
    // zero or at least two operands.
    classifier LoneUnion unions Wheel;
    classifier LoneIntersection intersects Wheel;
    classifier LoneDifference differences Wheel;

    // An unresolved operand stays an explicit unresolved outcome; it does not shrink the operand
    // set into a smaller one that would be silently well-formed.
    classifier Partial unions Wheel, Missing1, Missing2;

    // The feature owner goes through the same lowering.
    feature spokes;
    feature hubs;
    feature rims unions spokes, hubs;
}

// The set meanings entail type conformance without being specializations. Each feature below
// subsets one whose type is stated, so a conformance failure surfaces as
// `subsetting_type_incompatible` at the subsetting clause.
package Entailment {
    classifier Vehicle;
    classifier Car specializes Vehicle;
    classifier Van specializes Vehicle;
    classifier Boat;

    classifier RoadVehicle unions Car, Van;
    classifier CarAndVan intersects Car, Van;
    classifier NotVan differences Car, Van;

    feature anyVehicle : Vehicle;
    feature anyCar : Car;
    feature anyBoat : Boat;

    // Accepted. Every operand of the union is a `Vehicle`, so the union is one too.
    feature roadOnly : RoadVehicle subsets anyVehicle;
    // Rejected. Only one operand is a `Car`; a union is not included in what merely some operand
    // is included in.
    feature roadAsCar : RoadVehicle subsets anyCar;
    // Accepted. Each operand is included in the union it belongs to.
    feature carAsRoad : Car subsets roadOnly;

    // Accepted. An intersection is included in each of its operands, so it is a `Car`.
    feature bothAsCar : CarAndVan subsets anyCar;

    // Accepted: a difference is included in its first operand, which is a `Car`. Rejected: it has
    // no positive inclusion in what it excludes, and none in an unrelated type.
    feature reducedAsCar : NotVan subsets anyCar;
    feature reducedAsBoat : NotVan subsets anyBoat;
}
