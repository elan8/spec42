        port left : Signal;
        port right : Signal;
        connect left to right;
    }

        port left : Signal;
        port right : Power;
        connect left to right;
    }

    part def EndpointIsNotAPort {
        port left : Signal;
        part load : Load;
        connect left to load;
    }

    part def RepeatedPair {
        port left : Signal;
        port right : Signal;
        connect left to right;
        connect left to right;
    }

    part def NothingConnectsThisPort {
        port lonely : Signal;
    }
}
